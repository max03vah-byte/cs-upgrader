// --- ИСПРАВЛЕННАЯ ОТРИСОВКА КОЛЕСА --- function updateUI() { const chanceText = document.getElementById('chanceText'); const chanceLabel = document.getElementById('chanceLabel'); const chooseText = document.getElementById('chooseText'); const wheel = document.getElementById('wheel'); if (!selectedLeft || !selectedRight) { chanceText.style.display = 'none'; chanceLabel.style.display = 'none'; chooseText.style.display = 'block'; // Убираем from 180deg, чтобы все начиналось сверху wheel.style.background = 'conic-gradient(var(--accent) 0deg, var(--accent) 0deg, #0B0E14 0deg)'; return; } chanceText.style.display = 'block'; chanceLabel.style.display = 'block'; chooseText.style.display = 'none'; let chance = selectedLeft.price / selectedRight.price; if (chance > 1) chance = 1; chanceText.textContent = (chance * 100).toFixed(1) + '%'; const deg = chance * 360; // ИСПРАВЛЕНО: убрал from 180deg wheel.style.background = `conic-gradient(var(--accent) 0deg, var(--accent) ${deg}deg, #0B0E14 ${deg}deg)`; } // --- ИСПРАВЛЕННАЯ АНИМАЦИЯ КОЛЕСА --- document.getElementById('upgradeBtn').addEventListener('click', () => { if (isSpinning) return; if (!selectedLeft) return alert('Выберите скин для ставки!'); if (!selectedRight) return alert('Выберите цель!'); if (selectedLeft.price >= selectedRight.price) return alert('Цель должна быть дороже!'); const cheatVal = isAdmin ? parseInt(document.getElementById('cheatSlider').value) : 0; let diff = selectedRight.price - selectedLeft.price; let cheatCost = Math.floor((cheatVal / 100) * diff); if (state.balance < cheatCost) { return showToast('Недостаточно средств для оплаты подкрутки!'); } isSpinning = true; if (cheatCost > 0) { state.balance -= cheatCost; updateBalance(); } let baseChance = selectedLeft.price / selectedRight.price; if (baseChance > 1) baseChance = 1; let finalChance = Math.min(1, baseChance + 0.10); // Скрытая подкрутка if (cheatVal > 0) { finalChance = Math.min(1, finalChance + (cheatVal / 100)); } // Заранее определяем, выиграет игрок или нет const win = Math.random() < finalChance; const wonItem = selectedRight; // Удаляем ставку из инвентаря state.inventory.splice(selectedLeft.index, 1); saveState(); const arrow = document.getElementById('arrow'); arrow.style.transition = 'none'; arrow.style.transform = 'rotate(0deg)'; void arrow.offsetWidth; // --- РАСЧЕТ УГЛА --- // Угол выигрышной зоны (например, при 20% это 72 градуса) let winZoneDeg = baseChance * 360; // Если победа, стрелка падает от 0 до winZoneDeg // Если проигрыш, стрелка падает от winZoneDeg до 360 let targetAngle; if (win) { targetAngle = Math.random() * winZoneDeg; } else { let loseZoneDeg = 360 - winZoneDeg; targetAngle = winZoneDeg + (Math.random() * loseZoneDeg); } // 5 полных оборотов + угол let totalRotation = (360 * 5) + targetAngle; setTimeout(() => { arrow.style.transition = 'transform 4s cubic-bezier(0.1, 0.7, 0.1, 1)'; arrow.style.transform = `rotate(${totalRotation}deg)`; }, 50); document.getElementById('upgradeBtn').disabled = true; setTimeout(() => { const overlay = document.getElementById('resultOverlay'); const wheel = document.getElementById('wheel'); overlay.classList.add('show'); if (win) { state.inventory.push(wonItem); saveState(); document.getElementById('resultTitle').textContent = 'ПОБЕДА'; document.getElementById('resultTitle').style.color = 'var(--win)'; wheel.classList.add('win-border'); wheel.classList.remove('lose-border'); showToast('Вы выиграли!'); } else { document.getElementById('resultTitle').textContent = 'ПРОИГРЫШ'; document.getElementById('resultTitle').style.color = 'var(--lose)'; wheel.classList.add('lose-border'); wheel.classList.remove('win-border'); showToast('Вы проиграли!'); } document.getElementById('resultImg').src = wonItem.img; document.getElementById('resultName').textContent = wonItem.name; document.getElementById('resultPrice').textContent = formatMoney(wonItem.price); selectedLeft = null; document.getElementById('leftImg').classList.add('hidden'); document.getElementById('leftEmpty').classList.remove('hidden'); document.getElementById('leftName').textContent = ''; document.getElementById('leftPrice').textContent = ''; updateUI(); updateCheatCost(); document.getElementById('upgradeBtn').disabled = false; isSpinning = false; setTimeout(() => { overlay.classList.remove('show'); wheel.classList.remove('win-border', 'lose-border'); }, 3000); }, 4000); });
